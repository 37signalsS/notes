Я хочу создать кластер kubernetes при помощи kind на своем ноутбуке с 3 мастер нодами(etcd, kube api, sheduler, control manager), 2 обычные ноды на которых будет запущен nginx выводящий "Hello DEvops", 1 ingress нода. Я создал следующие конфиги, проверь их на правильность и на то можно ли с их помощью можно реализовать мою идею(файл index.html мной создан и добавлен в /data):
  ```
kind: Cluster
apiVersion: kind.x-k8s.io/v1alpha4
nodes:
  - role: control-plane
    extraPortMappings:
      - containerPort: 80
        hostPort: 8080
        protocol: TCP
  - role: control-plane
    kubeadmConfigPatches:
      - |
        kind: InitConfiguration
        nodeRegistration:
          kubeletExtraArgs:
            node-labels: "ingress-ready=true" 
  - role: control-plane
  - role: worker
  - role: worker

  
  
apiVersion: v1
kind: ConfigMap
metadata:
  name: my-config
data:
  app.conf: |
    server {
        listen 80;
        server_name localhost;
    
        location / {
            root /data;
            index index.html;
        }
    }
    
      
apiVersion: v1
kind: PersistentVolume
metadata:
  name: my-pv
  labels:
    type: local
spec:
  capacity:
    storage: 1Gi
  volumeMode: Filesystem
  accessModes:
    - ReadWriteOnce # или ReadWriteMany, если необходимо
  storageClassName: my-first-storage-class
  persistentVolumeReclaimPolicy: Delete
  local: # позволяет обращаться к Фс хостовой машины, не умеют в переезд
    path: /data
  nodeAffinity:
    required:
      nodeSelectorTerms:
        - matchExpressions:
            - key: kubernetes.io/hostname
              operator: In
              values:
                - my-cluster-worker  # имя узла к ФС которого мы хотим подключитьс
                
                           
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: my-pvc
spec:
  accessModes:
    - ReadWriteOnce # эксклюзивный доступ к диску, ReadWriteMany - множественный доступ к диску
  storageClassName: my-first-storage-class
  resources:
    requests:
      storage: 1Gi
      
      
apiVersion: v1
kind: Service
metadata:
  name: my-first-nginx-service
spec:
  selector:
    app: my-first-nginx
  ports:
    - protocol: TCP
      port: 80
      targetPort: 80
  type: ClusterIP
  
  
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: my-first-storage-class
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: WaitForFirstConsumer # создание связки PV and PVC только при появлении нагрузки


apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: my-first-ingress-nginx
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  rules:
    - host: localhost
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: my-first-nginx-service
                port:
                  number: 80
                  
                                   
apiVersion: apps/v1
kind: Deployment
metadata:
  name: my-first-nginx-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: my-first-nginx
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 1
  template:
    metadata:
      labels:
        app: my-first-nginx
    spec:
      containers:
        - name: my-container
          image: nginx:latest
          ports:
            - containerPort: 80
          volumeMounts:
            # Монтируем ConfigMap как том для конфигурационных файлов
            - name: config-volume
              mountPath: /etc/nginx/conf.d
            - name: data-volume
              mountPath: /data
      volumes:
        - name: config-volume
          configMap:
            name: my-config
        - name: data-volume
          persistentVolumeClaim:
            claimName: my-pvc
  ```
